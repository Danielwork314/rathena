//===== rAthena Script =======================================
//= Enchant Database Manager Frontend
//===== Description: =========================================
//= Always-visible frontend. The Zeny-only backend is isolated
//= so this NPC still appears and reports a useful error if the
//= custom item_enchant_zeny source command is not compiled.
//============================================================

prontera,156,168,4	script	Enchant Database Manager#custom_db	4_F_TELEPORTER,{
	mes "[Enchant Database Manager]";
	if (!$@EnchantDBManagerBackendReady) {
		mes "The NPC frontend loaded, but its Zeny-only enchant backend did not.";
		mes "Check the map-server startup or @reloadscript log for an unknown item_enchant_zeny command or an error in enchant_database_manager_backend.txt.";
		mes "The source patch must be present in the running map-server binary.";
		close;
	}
	mes "Open the official enchant database groups in custom Zeny-only mode.";
	mes "Materials are not checked or consumed. Official equipment, refine, grade, slot-order, chance, reset and upgrade rules remain active.";
	next;
	if (select("Open Enchant Database:Close") == 2)
		close;
	close2;
	doevent "EnchantDatabaseManagerBackend::OnOpen";
	end;

OnInit:
	$@EnchantDBManagerBackendReady = 0;
	end;
}
