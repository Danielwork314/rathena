//===== rAthena Script =======================================
//= Card Removal - Zeny Only, Guaranteed
//===== Description: =========================================
//= Removes every normal card from one equipped item for a flat
//= 50,000 Zeny. No materials, no failure, no destruction.
//= The equipment, refine/grade/enchant state, and all cards are preserved.
//============================================================

prontera,139,225,4	script	Card Removal	78,{
	.@fee = 50000;
	disable_items;
	mes "[Card Removal]";
	mes "I remove all cards from one equipped item.";
	mes "Fee: ^0055FF"+callfunc("F_InsertComma",.@fee)+" Zeny^000000";
	mes "Success rate: ^00AA00100%^000000";
	mes "No materials are required. The equipment and every removed card are preserved.";
	next;
	if (select("Remove cards:Cancel") == 2) close;

	setarray .@slots[0],EQI_HEAD_TOP,EQI_HEAD_MID,EQI_HEAD_LOW,EQI_ARMOR,EQI_HAND_L,EQI_HAND_R,EQI_GARMENT,EQI_SHOES,EQI_ACC_L,EQI_ACC_R;
	for (.@i = 0; .@i < getarraysize(.@slots); ++.@i) {
		.@slot = .@slots[.@i];
		if (getequipisequiped(.@slot))
			.@menu$ += F_getpositionname(.@slot)+" - ["+getequipname(.@slot)+"]:";
		else
			.@menu$ += "^777777"+F_getpositionname(.@slot)+" - Empty^000000:";
	}
	.@selected = select(.@menu$)-1;
	.@part = .@slots[.@selected];

	if (!getequipisequiped(.@part)) {
		mes "[Card Removal]";
		mes "There is no equipment in that position.";
		close;
	}
	.@card_count = 0;
	for (.@i = 0; .@i < MAX_SLOTS; ++.@i) {
		.@card_id = getequipcardid(.@part,.@i);
		if (.@card_id && getiteminfo(.@card_id,ITEMINFO_SUBTYPE) == CARD_NORMAL) {
			.@return_card[.@card_count] = .@card_id;
			.@return_count[.@card_count] = 1;
			++.@card_count;
		}
	}
	if (.@card_count <= 0) {
		mes "[Card Removal]";
		mes "That equipment has no removable cards.";
		close;
	}
	if (Zeny < .@fee) {
		mes "[Card Removal]";
		mes "You need "+callfunc("F_InsertComma",.@fee)+" Zeny.";
		close;
	}
	if (!checkweight2(.@return_card,.@return_count)) {
		mes "[Card Removal]";
		mes "Your inventory does not have enough weight or item capacity to receive every removed card.";
		close;
	}

	mes "[Card Removal]";
	mes "Equipment: ^0055FF"+getequipname(.@part)+"^000000";
	mes "Cards returned: ^0055FF"+.@card_count+"^000000";
	mes "Cost: ^0055FF"+callfunc("F_InsertComma",.@fee)+" Zeny^000000";
	mes "The item, refine, grade, enchantments, and cards will all be preserved.";
	next;
	if (select("Confirm removal:Cancel") == 2) close;
	if (Zeny < .@fee) {
		mes "[Card Removal]";
		mes "Your Zeny changed before confirmation.";
		close;
	}

	Zeny -= .@fee;
	successremovecards .@part;
	specialeffect2 EF_MAXPOWER;
	mes "[Card Removal]";
	mes "Removal completed successfully.";
	mes "Your equipment and all cards were preserved.";
	close;

OnInit:
	// Disable the official material/failure-based separation NPCs.
	disablenpc "Jeremy#pa0829";
	disablenpc "Richard#pa0829";
	end;
}
