function	script	F_MvpArenaKeeper	{
	.@suffix = getarg(0);
	.@title$ = getarg(1);
	mes "[" + .@title$ + "]";
	mes "Choose an arena copy.";
	for (.@i = 1; .@i <= 8; ++.@i)
		.@menu$ += "Arena " + .@i + " [" + getmapusers("pvp_n_"+.@i+"-"+.@suffix) + "/20]:";
	.@room = select(.@menu$);
	.@map$ = "pvp_n_"+.@room+"-"+.@suffix;
	if (getmapusers(.@map$) >= 20) {
		mes "[" + .@title$ + "]";
		mes "That arena is full.";
		close;
	}
	warp .@map$,102,102;
	close;
}

function	script	F_MvpArenaSelect	{
	.@group = getarg(0);
	switch (.@group) {
	case 1: .@array$ = "$@MVP_ARENA_L1_99"; .@size = $@MVP_ARENA_L1_99_SIZE; break;
	case 2: .@array$ = "$@MVP_ARENA_L100_149"; .@size = $@MVP_ARENA_L100_149_SIZE; break;
	case 3: .@array$ = "$@MVP_ARENA_L150_199"; .@size = $@MVP_ARENA_L150_199_SIZE; break;
	case 4: .@array$ = "$@MVP_ARENA_L200_PLUS"; .@size = $@MVP_ARENA_L200_PLUS_SIZE; break;
	default: return 0;
	}

	.@per_page = 6;
	.@page = 0;
	while (1) {
		.@start = .@page * .@per_page;
		.@remaining = .@size - .@start;
		.@shown = .@remaining > .@per_page ? .@per_page : .@remaining;
		.@menu$ = "";
		for (.@i = 0; .@i < .@shown; ++.@i) {
			.@id = getd(.@array$ + "[" + (.@start + .@i) + "]");
			.@menu$ += getmonsterinfo(.@id,MOB_NAME) + " [Lv " + getmonsterinfo(.@id,MOB_LV) + "]:";
		}
		.@has_prev = .@page > 0;
		.@has_next = (.@start + .@shown) < .@size;
		if (.@has_prev) .@menu$ += "^777777Previous page^000000:";
		if (.@has_next) .@menu$ += "^0055FFNext page^000000:";
		.@menu$ += "Cancel";
		.@choice = select(.@menu$) - 1;
		if (.@choice < .@shown)
			return getd(.@array$ + "[" + (.@start + .@choice) + "]");
		.@nav = .@choice - .@shown;
		if (.@has_prev) {
			if (.@nav == 0) { --.@page; continue; }
			--.@nav;
		}
		if (.@has_next && .@nav == 0) { ++.@page; continue; }
		return 0;
	}
}

function	script	F_MvpArenaRoom	{
	.@group = getarg(0);
	mes "[MVP Arena Summoner]";
	mes "Only one MVP can be active in this room.";
	next;
	switch(select("Summon an MVP:Heal:Return to lobby:Return to Prontera:Cancel")) {
	case 1:
		if (mobcount("this",strnpcinfo(3)+"::OnMvpDead") > 0) {
			mes "[MVP Arena Summoner]";
			mes "Defeat the active MVP before summoning another one.";
			close;
		}
		.@mob_id = callfunc("F_MvpArenaSelect",.@group);
		if (!.@mob_id) close;
		mes "[MVP Arena Summoner]";
		mes "Summon ^FF0000"+getmonsterinfo(.@mob_id,MOB_NAME)+"^000000?";
		mes "Level: "+getmonsterinfo(.@mob_id,MOB_LV);
		next;
		if (select("Summon:Cancel") == 2) close;
		close2;
		monster "this",0,0,getmonsterinfo(.@mob_id,MOB_NAME),.@mob_id,1,strnpcinfo(3)+"::OnMvpDead";
		mapannounce strnpcinfo(4),getmonsterinfo(.@mob_id,MOB_NAME)+" has been summoned!",bc_map|bc_red;
		end;
	case 2:
		specialeffect2 EF_HEAL2;
		percentheal 100,100;
		close;
	case 3:
		warp "quiz_00",50,24;
		close;
	case 4:
		warp "prontera",156,179;
		close;
	default:
		close;
	}
}
