//============================================================

function	script	F_MvpArenaSelect	{
	.@start = getarg(0);
	.@size = getarg(1);
	.@title$ = getarg(2);
	.@per_page = 6;
	.@page = 0;

	while (1) {
		.@offset = .@page * .@per_page;
		.@remaining = .@size - .@offset;
		.@shown = (.@remaining > .@per_page) ? .@per_page : .@remaining;
		.@menu$ = "";

		for (.@i = 0; .@i < .@shown; ++.@i) {
			.@index = .@start + .@offset + .@i;
			.@mob_id = $@MVP_ARENA_ID[.@index];
			.@menu$ += getmonsterinfo(.@mob_id,MOB_NAME) + " [Lv " + getmonsterinfo(.@mob_id,MOB_LV) + "]:";
		}

		.@has_prev = (.@page > 0);
		.@has_next = ((.@offset + .@shown) < .@size);
		if (.@has_prev)
			.@menu$ += "^777777Previous page^000000:";
		if (.@has_next)
			.@menu$ += "^0055FFNext page^000000:";
		.@menu$ += "Cancel";

		clear;
		mes "[ " + .@title$ + " ]";
		mes "Choose an MVP room.";
		.@choice = select(.@menu$) - 1;

		if (.@choice < .@shown)
			return .@start + .@offset + .@choice;

		.@nav = .@choice - .@shown;
		if (.@has_prev) {
			if (.@nav == 0) {
				--.@page;
				continue;
			}
			--.@nav;
		}
		if (.@has_next && .@nav == 0) {
			++.@page;
			continue;
		}
		return -1;
	}
}

// Spawn one cardinal slot directly through a stable global event label.
// This avoids calling an event on a freshly duplicated instance NPC.
function	script	F_MvpArenaEnsureSlot	{
	.@instance_id = getarg(0);
	.@mob_id = getarg(1);
	.@slot = getarg(2);
	.@map$ = instance_mapname("pvp_y_2-5",.@instance_id);
	if (.@map$ == "" || .@mob_id <= 0)
		return 0;

	.@event$ = "MVP_Arena_Runtime::OnMvpDead" + .@slot;
	if (mobcount(.@map$,.@event$) > 0)
		return 1;

	switch (.@slot) {
	case 1: // East
		.@x = 260;
		.@y = 160;
		break;
	case 2: // South
		.@x = 160;
		.@y = 35;
		break;
	case 3: // West
		.@x = 60;
		.@y = 160;
		break;
	case 4: // North
		.@x = 160;
		.@y = 283;
		break;
	default:
		return 0;
	}

	monster .@map$,.@x,.@y,"--ja--",.@mob_id,1,.@event$;
	return mobcount(.@map$,.@event$) > 0;
}

function	script	F_MvpArenaEnsureRoom	{
	.@instance_id = getarg(0);
	.@mob_id = getarg(1);
	.@ready = 1;
	for (.@slot = 1; .@slot <= 4; ++.@slot)
		.@ready = callfunc("F_MvpArenaEnsureSlot",.@instance_id,.@mob_id,.@slot) && .@ready;
	return .@ready;
}

// Resolve the instance that owns a unique instance-map name.
function	script	F_MvpArenaFindInstanceByMap	{
	.@target$ = getarg(0);
	.@count = instance_list("pvp_y_2-5",IM_NONE);
	for (.@i = 0; .@i < .@count; ++.@i) {
		.@instance_id = .@instance_list[.@i];
		if (instance_mapname("pvp_y_2-5",.@instance_id) == .@target$)
			return .@instance_id;
	}
	return 0;
}

function	script	F_MvpArenaEnter	{
	.@index = getarg(0);
	if (.@index < 0 || .@index >= $@MVP_ARENA_SIZE)
		return 0;

	.@mob_id = $@MVP_ARENA_ID[.@index];
	.@instance_id = $@MVP_ARENA_INSTANCE[.@index];

	// Discard stale mappings left by idle destruction or script reload.
	if (.@instance_id > 0 && instance_live_info(ILI_NAME,.@instance_id) != "Complete MVP Arena Room") {
		$@MVP_ARENA_INSTANCE[.@index] = 0;
		.@instance_id = 0;
	}

	if (.@instance_id <= 0) {
		.@instance_id = instance_create("Complete MVP Arena Room",IM_NONE);
		if (.@instance_id < 1) {
			mes "[ MVP Arena ]";
			mes "The room could not be created.";
			mes "Instance error: " + .@instance_id;
			close;
		}

		$@MVP_ARENA_INSTANCE[.@index] = .@instance_id;
	}

	// Instance maps are created through rAthena's queue. Do not access
	// instance variables until instance_mapname confirms the room is busy.
	.@map$ = instance_mapname("pvp_y_2-5",.@instance_id);
	if (.@map$ == "") {
		mes "[ MVP Arena ]";
		mes "The " + getmonsterinfo(.@mob_id,MOB_NAME) + " room is being prepared.";
		mes "Please choose the same room again after the instance queue finishes.";
		close;
	}

	// Assign the canonical room owner only after the instance is fully ready.
	if (getinstancevar('mvp_id,.@instance_id) <= 0) {
		setinstancevar('mvp_id,.@mob_id,.@instance_id);
		setinstancevar('room_index,.@index + 1,.@instance_id);
	}

	if (getinstancevar('mvp_id,.@instance_id) != .@mob_id) {
		mes "[ MVP Arena ]";
		mes "Room ownership validation failed.";
		mes "Please report MVP ID " + .@mob_id + ".";
		close;
	}

	if (!getinstancevar('configured,.@instance_id)) {
		if (!callfunc("F_MvpArenaEnsureRoom",.@instance_id,.@mob_id)) {
			mes "[ MVP Arena ]";
			mes "The room failed to initialize its four MVP slots.";
			mes "Please report MVP ID " + .@mob_id + ".";
			close;
		}
		setinstancevar('configured,1,.@instance_id);
	}

	if (getmapusers(.@map$) >= 20) {
		mes "[ MVP Arena ]";
		mes "This MVP room is full.";
		close;
	}

	close2;
	warp .@map$,160,160;
	end;
}

function	script	F_MvpArenaKeeper	{
	.@start = getarg(0);
	.@size = getarg(1);
	.@title$ = getarg(2);
	.@index = callfunc("F_MvpArenaSelect",.@start,.@size,.@title$);
	if (.@index < 0)
		close;

	.@mob_id = $@MVP_ARENA_ID[.@index];
	mes "[ " + .@title$ + " ]";
	mes "Enter the dedicated room for";
	mes "^FF0000" + getmonsterinfo(.@mob_id,MOB_NAME) + "^000000?";
	mes "Monster ID: " + .@mob_id;
	mes "Level: " + getmonsterinfo(.@mob_id,MOB_LV);
	mes "Room population: 4 identical MVPs, spread far apart";
	mes "Respawn: exactly 20 seconds";
	next;
	if (select("Enter room:Cancel") == 2)
		close;
	callfunc "F_MvpArenaEnter",.@index;
	close;
}

// Stable global death-event owner. Monster death labels always exist independently
// of instance NPC duplication timing. The killer RID supplies the unique room map.
-	script	MVP_Arena_Runtime	-1,{
S_Respawn:
	.@slot = getarg(0);
	.@map$ = strcharinfo(3);
	detachrid;
	sleep 20000;

	.@instance_id = callfunc("F_MvpArenaFindInstanceByMap",.@map$);
	if (.@instance_id <= 0)
		return;
	.@mob_id = getinstancevar('mvp_id,.@instance_id);
	if (.@mob_id <= 0 || !getinstancevar('configured,.@instance_id))
		return;
	callfunc "F_MvpArenaEnsureSlot",.@instance_id,.@mob_id,.@slot;
	return;

OnMvpDead1:
	callsub S_Respawn,1;
	end;
OnMvpDead2:
	callsub S_Respawn,2;
	end;
OnMvpDead3:
	callsub S_Respawn,3;
	end;
OnMvpDead4:
	callsub S_Respawn,4;
	end;
}
