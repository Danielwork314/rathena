//= Eight private rooms reserved for a party, guild, or account.
//= Entry and summoning are free. After selecting a monster, players
//= may summon it once, ten times total, or continuously. Repeated modes
//= respawn the monster 20 seconds after each defeat.

function	script	F_MvpRoomSelectMonster	{
	.@start = getarg(0);
	.@size = getarg(1);
	.@title$ = getarg(2);

	mes "[" + .@title$ + "]";
	mes "Choose a level range.";
	next;
	.@category = select(
		"Level 1-99",
		"Level 100-149",
		"Level 150-199",
		"Level 200-249",
		"Level 250+",
		"All levels",
		"Cancel"
	);
	if (.@category == 7)
		return 0;

	switch (.@category) {
	case 1:
		.@minimum = 1;
		.@maximum = 99;
		break;
	case 2:
		.@minimum = 100;
		.@maximum = 149;
		break;
	case 3:
		.@minimum = 150;
		.@maximum = 199;
		break;
	case 4:
		.@minimum = 200;
		.@maximum = 249;
		break;
	case 5:
		.@minimum = 250;
		.@maximum = 9999;
		break;
	default:
		.@minimum = 1;
		.@maximum = 9999;
		break;
	}

	deletearray .@matches[0], getarraysize(.@matches);
	.@match_count = 0;
	for (.@i = 0; .@i < .@size; ++.@i) {
		.@mob_id = $@MVP_ROOM_MONSTER_ID[.@start + .@i];
		.@level = getmonsterinfo(.@mob_id, MOB_LV);
		if (.@level >= .@minimum && .@level <= .@maximum)
			.@matches[.@match_count++] = .@mob_id;
	}

	if (!.@match_count) {
		mes "[" + .@title$ + "]";
		mes "There are no validated monsters in this level range.";
		return 0;
	}

	.@per_page = 6;
	.@page = 0;
	while (1) {
		.@offset = .@page * .@per_page;
		.@remaining = .@match_count - .@offset;
		.@shown = (.@remaining > .@per_page) ? .@per_page : .@remaining;
		.@menu$ = "";

		for (.@i = 0; .@i < .@shown; ++.@i) {
			.@mob_id = .@matches[.@offset + .@i];
			.@menu$ += getmonsterinfo(.@mob_id, MOB_NAME)
				+ " [Lv " + getmonsterinfo(.@mob_id, MOB_LV)
				+ " / ID " + .@mob_id + "]:";
		}

		.@has_previous = (.@page > 0);
		.@has_next = ((.@offset + .@shown) < .@match_count);
		if (.@has_previous)
			.@menu$ += "^777777Previous page^000000:";
		if (.@has_next)
			.@menu$ += "^0055FFNext page^000000:";
		.@menu$ += "Cancel";

		clear;
		mes "[" + .@title$ + "]";
		mes "Choose the monster to summon.";
		mes "You will choose the summon count after selecting it.";
		.@choice = select(.@menu$) - 1;

		if (.@choice < .@shown)
			return .@matches[.@offset + .@choice];

		.@navigation = .@choice - .@shown;
		if (.@has_previous) {
			if (.@navigation == 0) {
				--.@page;
				continue;
			}
			--.@navigation;
		}
		if (.@has_next && .@navigation == 0) {
			++.@page;
			continue;
		}
		return 0;
	}
}

	mes "Each reservation lasts "
		+ getvariableofnpc(.timeout, "MVP Summoner") + " minutes.";
	mes " ";
	for (.@i = 1; .@i <= 8; ++.@i)
		if (getvariableofnpc(.renttime[.@i], "MVP Summoner"))
			mes "Room #" + .@i + " = " + .color$[.type[.@i]]
				+ .whoinuse$[.@i] + "^000000";
	next;

	.@room = select(
		"MVP Room 1 [" + getmapusers("06guild_01") + "]",
		"MVP Room 2 [" + getmapusers("06guild_02") + "]",
		"MVP Room 3 [" + getmapusers("06guild_03") + "]",
		"MVP Room 4 [" + getmapusers("06guild_04") + "]",
		"MVP Room 5 [" + getmapusers("06guild_05") + "]",
		"MVP Room 6 [" + getmapusers("06guild_06") + "]",
		"MVP Room 7 [" + getmapusers("06guild_07") + "]",
		"MVP Room 8 [" + getmapusers("06guild_08") + "]"
	);

		if (.inuseid[.@room] == getcharid(.type[.@room])) {
			warp "06guild_0" + .@room, 0, 0;
			close;
		}
		mes "[Private MVP Room]";
		mes "This room is reserved for";
		mes .color$[.type[.@room]] + .whoinuse$[.@room] + "^000000.";
		mes "Please select another room.";
		close;
	}

	mes "[Private MVP Room]";
	mes "Reserve this free room for...";
	next;
	.@type = select(
		"For my party members",
		"For my guild members",
		"For personal account use"
	);

	if (!getcharid(.@type)) {
		mes "[Private MVP Room]";
		mes "You do not own a " + ((.@type == 1) ? "Party" : "Guild") + ".";
		close;
	}
	if (getvariableofnpc(.renttime[.@room], "MVP Summoner")) {
		mes "[Private MVP Room]";
		mes "Somebody else registered this room first.";
		close;
	}

	for (.@i = 1; .@i <= 8; ++.@i) {
		if (getvariableofnpc(.renttime[.@i], "MVP Summoner")
			&& .@type == .type[.@i]
			&& getcharid(.@type) == .inuseid[.@i]) {
			mes "[Private MVP Room]";
			mes "You already reserved Room #" + .@i + ".";
			close;
		}
	}

	.type[.@room] = .@type;
	.inuseid[.@room] = getcharid(.@type);
	.whoinuse$[.@room] = strcharinfo((.@type == 3) ? 0 : .@type);

	killmonsterall "06guild_0" + .@room;
	warp "06guild_0" + .@room, 0, 0;
	donpcevent "MVP Summoner#" + .@room + "::OnEnterMap";
	close;

OnInit:
	.color$[1] = "^EE8800";
	.color$[2] = "^70CC11";
	.color$[3] = "^0000FF";
	end;
}
